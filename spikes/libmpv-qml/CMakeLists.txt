cmake_minimum_required(VERSION 3.21)
project(mpvspike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(MpvQt REQUIRED)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(mpvspike main.cpp mpvitem.cpp mpvitem.h)
qt_add_qml_module(mpvspike URI Spike VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(mpvspike PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick MpvQt::MpvQt)
